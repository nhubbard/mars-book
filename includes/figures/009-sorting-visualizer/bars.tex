  \newcommand{\SortBars}[3]{%
    \begin{scope}[xshift=#1]
      \foreach \v [count=\x from 0] in {#2} {
          \fill[black!45] (\x-0.5,2.5-\v) rectangle (\x+0.5,7.5);
          \node[below] at (\x,7.7) {\scriptsize $\v$};
        }
      \foreach \k in {0,...,8} {
          \draw[gray] (\k-0.5,-0.5)--(\k-0.5,7.5);
          \draw[gray] (-0.5,\k-0.5)--(7.5,\k-0.5);
        }
      \node[above] at (3.5,-0.8) {#3};
    \end{scope}}
  \begin{tikzpicture}[alt={Three bar grids: original values 2,−2,3,−4,0,1,−3,−1; first adjacent exchange −2,2,3,−4,0,1,−3,−1; final sorted −4,−3,−2,−1,0,1,2,3.},x=0.32cm,y=-0.32cm]
    \SortBars{0cm}{2,-2,3,-4,0,1,-3,-1}{Initial}
    \SortBars{3.5cm}{-2,2,3,-4,0,1,-3,-1}{First insertion exchange}
    \SortBars{7cm}{-4,-3,-2,-1,0,1,2,3}{Final}
