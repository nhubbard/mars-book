\section{Check the language before writing the loops}\label{expression-questions}
\begin{MARSExercise}{Precedence and association}
  Evaluate \texttt{9-6/2-1}. State the successive reductions and the final value.
  Which interpretation follows the grammar?
  \begin{itemize}
    \item \textbf{A} Reduce \texttt{9-6} before dividing.
    \item \textbf{B} Reduce \texttt{6/2}, then perform the two subtractions left to right.
    \item \textbf{C} Reduce the rightmost subtraction before the leftmost one.
  \end{itemize}
\end{MARSExercise}
\begin{MARSExercise}{Error position}
  For \texttt{1/(2-2}, choose the required status and offset.
  \begin{itemize}
    \item \textbf{A} Division by zero at byte one.
    \item \textbf{B} Syntax error at byte six.
    \item \textbf{C} Success with value zero.
  \end{itemize}
  Explain which parser call cannot yet return a successful value.
\end{MARSExercise}
\begin{MARSExercise}{Intermediate range}
  Explain why \texttt{(2147483647+1)*0} fails even though simplifying the mathematical
  expression gives zero. Identify the operator byte that must be reported. Then give
  a valid expression for the minimum signed word and a unary expression that overflows
  when applied to that value.
\end{MARSExercise}
