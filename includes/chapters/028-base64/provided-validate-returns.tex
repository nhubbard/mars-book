\section{Supplied validate returns}
Retain this support code while replacing the named arithmetic regions. Its normal and development returns restore exactly the frame allocated by that routine. Status four reports unfinished whole-function work; it is distinct from the complete error results.

\Needspace{27\baselineskip}
\MARSCodeLabel{MIPS assembly}
\begin{minted}[samepage=true]{mars-idea}
validate_good:
    move $v0, $zero
    move $v1, $s2
    b validate_restore
    nop
validate_bad:
    li $v0, 2
    move $v1, $zero
    b validate_restore
    nop
validate_development:
    li $v0, 4
    move $v1, $zero
validate_restore:
    ld $s0, 0($sp)
    ld $s1, 8($sp)
    ld $s2, 16($sp)
    ld $s3, 24($sp)
    ld $ra, 32($sp)
    daddiu $sp, $sp, 48
    jr $ra
    nop
\end{minted}
