\section{Retain the source and occupancy across compression}
The supplied update frame saves four working registers and the return address in 48
bytes, including eight bytes of padding. It keeps the context in \texttt{s0}, source
walker in \texttt{s1}, remaining input in \texttt{s2} and buffer occupancy in
\texttt{s3}. Before the body starts, occupancy is loaded from the initialized context.
The common return writes its final working occupancy back to that context.

An incomplete update can temporarily return with occupancy sixty-four after filling
its first block. That is a development observation, outside the completed interface's
zero-through-sixty-three invariant. Prepare a fresh recipe image for each experiment;
do not continue such a context as though the update had completed. The complete resume
region resets occupancy after compression and restores the normal invariant.

\Needspace{15\baselineskip}
\MARSCodeLabel{Supplied entry and frame}
\begin{minted}[samepage=true]{mars-idea}
sha256_update:
    daddiu $sp, $sp, -48
    sd $s0, 0($sp)
    sd $s1, 8($sp)
    sd $s2, 16($sp)
    sd $s3, 24($sp)
    sd $ra, 32($sp)
    move $s0, $a0
    move $s1, $a1
    move $s2, $a2
    lwu $s3, 40($s0)
\end{minted}

\Needspace{22\baselineskip}
\MARSCodeLabel{Supplied normal and development exits}
\begin{minted}[samepage=true]{mars-idea}
update_done:
update_success:
    move $v0, $zero
    b update_restore
    nop
update_development:
    li $v0, 3
update_restore:
    sw $s3, 40($s0)
    ld $s0, 0($sp)
    ld $s1, 8($sp)
    ld $s2, 16($sp)
    ld $s3, 24($sp)
    ld $ra, 32($sp)
    daddiu $sp, $sp, 48
    jr $ra
    nop

\end{minted}
