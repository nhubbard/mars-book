% Licensed Minion fonts are installed locally, never bundled with the source.
% MARS_FONT_DIR overrides the default macOS user font library.
\edef\MARSFontDirectory{\directlua{tex.sprint(os.getenv('MARS_FONT_DIR') or (os.getenv('HOME') .. '/Library/Fonts/'))}}
\setmainfont{MinionPro-Regular.otf}[
  Path=\MARSFontDirectory,
  ItalicFont=MinionPro-It.otf,
  BoldFont=MinionPro-Bold.otf,
  BoldItalicFont=MinionPro-BoldIt.otf,
  Ligatures=TeX
]
\setmonofont{JetBrainsMono-Regular.otf}[
  ItalicFont=JetBrainsMono-Italic.otf,
  BoldFont=JetBrainsMono-Bold.otf,
  BoldItalicFont=JetBrainsMono-BoldItalic.otf,
  Scale=MatchLowercase,
  Ligatures={NoCommon,NoDiscretionary},
  RawFeature=-calt
]
\setmathfont{NotoSansMath-Regular.otf}[Path=\MARSFontDirectory]
