\section{Supplied power returns}
Retain this support code while replacing the marked arithmetic regions. Its saved-register slots and return paths are part of the cumulative starter. A framed development return reports status three; a normal return preserves the computed status while restoring the caller.

\Needspace{17\baselineskip}
\MARSCodeLabel{Supplied support code}
\begin{minted}[samepage=true]{mars-idea}
    move $v0, $zero
    b pow128_restore
    nop
pow128_development:
    li $v0, 3
pow128_restore:
    ld $s0, 64($sp)
    ld $s1, 72($sp)
    ld $ra, 80($sp)
    daddiu $sp, $sp, 96
    jr $ra
    nop
\end{minted}
