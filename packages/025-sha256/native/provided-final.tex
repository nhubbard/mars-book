\section{Keep finalization's saved length separate from its buffer}
Finalization's supplied frame retains the context, output pointer, working occupancy
and full bit count across compression calls. It occupies 48 bytes with the same
preserved-register layout as update. The context's original-byte count is read by the
first authored region; \texttt{s3} then retains its shifted bit count.

A partial finalizer may already have written padding or compressed a first padding
block before returning development status three. The context is consumed by that
experiment. Use the separately prepared recipe source again for another run. The
completed interface likewise requires initialization before a new message; it does
not define repeated finalization of an already finalized context.

\Needspace{14\baselineskip}
\MARSCodeLabel{Supplied entry and frame}
\begin{minted}[samepage=true]{mars-idea}
sha256_final:
    daddiu $sp, $sp, -48
    sd $s0, 0($sp)
    sd $s1, 8($sp)
    sd $s2, 16($sp)
    sd $s3, 24($sp)
    sd $ra, 32($sp)
    move $s0, $a0
    move $s1, $a1
    lwu $s2, 40($s0)
\end{minted}

\Needspace{20\baselineskip}
\MARSCodeLabel{Supplied normal and development exits}
\begin{minted}[samepage=true]{mars-idea}
final_done:
    move $v0, $zero
    b final_restore
    nop
final_development:
    li $v0, 3
final_restore:
    ld $s0, 0($sp)
    ld $s1, 8($sp)
    ld $s2, 16($sp)
    ld $s3, 24($sp)
    ld $ra, 32($sp)
    daddiu $sp, $sp, 48
    jr $ra
    nop

\end{minted}
