\section{Supplied decode returns}
Retain this support code while replacing the named arithmetic regions. Its normal and development returns restore exactly the frame allocated by that routine. Status four reports unfinished whole-function work; it is distinct from the complete error results.

\Needspace{31\baselineskip}
\MARSCodeLabel{MIPS assembly}
\begin{minted}[samepage=true]{mars-idea}
decode_done:
    move $v0, $zero
    move $v1, $s3
    b decode_restore
    nop
decode_capacity:
    li $v0, 3
decode_failed:
    move $v1, $zero
    b decode_restore
    nop
decode_development:
    li $v0, 4
    move $v1, $zero
decode_restore:
    ld $s0, 0($sp)
    ld $s1, 8($sp)
    ld $s2, 16($sp)
    ld $s3, 24($sp)
    ld $s4, 32($sp)
    ld $s5, 40($sp)
    ld $s6, 48($sp)
    ld $ra, 56($sp)
    daddiu $sp, $sp, 80
    jr $ra
    nop
\end{minted}
